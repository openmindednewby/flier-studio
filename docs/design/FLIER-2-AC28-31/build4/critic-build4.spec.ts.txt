async function aiMeasure(page: Page) {
  return page.evaluate(() => {
    const p = document.querySelector('[data-testid="all-ai-line"]') as HTMLElement; const a = p.querySelector('a')!;
    const r = (e: Element) => { const b = e.getBoundingClientRect(); return { x: Math.round(b.left), y: Math.round(b.top * 10) / 10, w: Math.round(b.width), h: Math.round(b.height * 10) / 10 }; };
    const rects = [...a.getClientRects()].map((b) => ({ y: Math.round(b.top), h: Math.round(b.height), w: Math.round(b.width) }));
    const prev = p.previousElementSibling, next = p.nextElementSibling;
    const ab = a.getBoundingClientRect(); const cx = ab.left + Math.min(ab.width / 2, 20);
    const hit = [ab.top + 2, ab.top + 8, ab.bottom - 8, ab.bottom - 2].map((y) => { const e = document.elementFromPoint(cx, y); return e === a || a.contains(e!) ? 'a' : `${e?.tagName}.${(e as HTMLElement)?.className}`; });
    return { p: r(p), a: r(a), rects, lineHeight: getComputedStyle(p).lineHeight, prev: prev ? r(prev) : null, next: next ? r(next) : null, gapAbove: prev ? Math.round(p.getBoundingClientRect().top - prev.getBoundingClientRect().bottom) : null, gapBelow: next ? Math.round(next.getBoundingClientRect().top - p.getBoundingClientRect().bottom) : null, hit, hscroll: document.documentElement.scrollWidth > innerWidth };
  });
}
async function markMeasure(page: Page) {
  return page.evaluate(() => [...document.querySelectorAll('.replace-mark')].map((m) => {
    const r = m.getBoundingClientRect(); const b = m.querySelector('b')!.getBoundingClientRect(); const t = m.closest('.trim')?.getBoundingClientRect();
    const ox = Math.max(0, Math.min(r.right, b.right) - Math.max(r.left, b.left)); const oy = Math.max(0, Math.min(r.bottom, b.bottom) - Math.max(r.top, b.top));
    return { n: m.textContent, inside: m.classList.contains('badge-inside'), off: m.classList.contains('off'), mark: { x: Math.round(r.left), y: Math.round(r.top), w: Math.round(r.width), h: Math.round(r.height) }, badge: { x: Math.round(b.left), w: Math.round(b.width), h: Math.round(b.height) }, overlapPx: { x: Math.round(ox), y: Math.round(oy) }, coverPct: Math.round((ox * oy) / (r.width * r.height) * 100), trimLeft: t ? Math.round(t.left) : null, badgeOffScreen: b.left < 0 || (t ? b.left < t.left - 0.5 : false) };
  }));
}
